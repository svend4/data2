## ЧАСТЬ I: 4 МЕТОДА → 4 СТРАТЕГИИ МАКСИМИЗАЦИИ ЛЗП

```python
import math
from typing import Callable, List, Tuple, Dict

class OptimizationETD:
    """
    4 метода оптимизации для максимизации ЛЗП системы 4 сфер.
    """

    # МЕТОД ЛАГРАНЖА: max ЛЗП при ограничениях
    @staticmethod
    def lagrange_etd(constraint_energy: float = 1.0) -> dict:
        """
        Метод множителей Лагранжа:
        max ЛЗП(МВС, СВС, БВС, ЧВС)
        при ограничении: МВС² + СВС² + БВС² + ЧВС² = E (энергия = const).

        Лагранжиан: L = ЛЗП − λ·(МВС²+СВС²+БВС²+ЧВС² − E).
        ∂L/∂МВС = 0, ∂L/∂СВС = 0, ... → условия экстремума.

        Решение: МВС = СВС = БВС = ЧВС = √(E/4).
        При E = π²/4: каждая сфера = π/4!
        = π/4-СИНХРОНИЗАЦИЯ = ОПТИМУМ ПО ЛАГРАНЖУ!
        """
        E = constraint_energy
        optimal_each = math.sqrt(E / 4)   # = √(E)/2
        optimal_lci = 2 * math.sqrt(E)    # = ||вектор|| = √(4·E/4·4) = 2√E

        # При E = (π/2)² = π²/4:
        E_special = (math.pi/2)**2
        optimal_each_special = math.sqrt(E_special / 4)
        # = π/(2·2) = π/4 ✓

        return {
            'метод': 'Лагранж',
            'условие': f'МВС²+СВС²+БВС²+ЧВС² = {E:.3f}',
            'оптимум': f'МВС=СВС=БВС=ЧВС = {optimal_each:.4f}',
            'максимальный_ЛЗП': optimal_lci,
            'при_E=π²/4': f'каждая сфера = π/4 = {math.pi/4:.4f}',
            'вывод': 'π/4-синхронизация = глобальный оптимум Лагранжа!'
        }

    # СИМПЛЕКС-МЕТОД: линейная оптимизация ЛЗП
    @staticmethod
    def simplex_etd(coefficients: List[float],
                     constraints: List[List[float]],
                     rhs: List[float]) -> dict:
        """
        Симплекс-метод для линейного ЛЗП:
        max c₁·МВС + c₂·СВС + c₃·БВС + c₄·ЧВС
        при: A·x ≤ b, x ≥ 0.

        Для ЕТД: максимизировать вклад каждой сферы
        при ограничениях на время и энергию.

        Симплекс: оптимум всегда в вершине многогранника допустимых решений.
        В задаче 4 сфер: вершины = 4 базисных состояния.
        """
        # Упрощённая версия: 4 переменных, равные коэффициенты
        n_vars = len(coefficients)
        # Оптимальный вектор = одна активная сфера при линейной задаче
        max_idx = max(range(n_vars), key=lambda i: coefficients[i])
        max_coef = coefficients[max_idx]

        solution = [0.0] * n_vars
        if rhs:
            solution[max_idx] = min(rhs) / max(constraints[0]) if constraints else 1.0

        return {
            'метод': 'Симплекс',
            'оптимальная_сфера': ['МВС', 'СВС', 'БВС', 'ЧВС'][max_idx],
            'оптимальный_ЛЗП': max_coef * solution[max_idx],
            'вывод': 'Линейная задача: оптимум — одна доминирующая сфера',
        }

    # ГРАДИЕНТНЫЙ СПУСК: итеративная оптимизация
    @staticmethod
    def gradient_descent_etd(initial_angles: List[float],
                               learning_rate: float = 0.1,
                               n_steps: int = 100) -> dict:
        """
        Градиентный спуск для нахождения π/4-оптимума:
        θₙ₊₁ = θₙ − α·∇Loss(θₙ).

        Loss = (ЛЗП - ЛЗП_max)² — отклонение от максимума.
        ∇Loss = -2·(ЛЗП_max - ЛЗП)·∂ЛЗП/∂θ.

        Цель: θ* = π/4 для каждой сферы (оптимум).
        Сходимость: θₙ → π/4 при α = O(1/n).
        """
        target = math.pi / 4   # целевой угол
        angles = list(initial_angles)
        history = [angles[:]]

        for step in range(n_steps):
            # Градиент: потери = Σ(θᵢ - π/4)²
            gradients = [2 * (a - target) for a in angles]
            # Шаг градиентного спуска
            angles = [a - learning_rate * g for a, g in zip(angles, gradients)]
            history.append(angles[:])

        final_lci = math.sqrt(sum(a**2 for a in angles))
        optimal_lci = math.sqrt(4 * target**2)

        return {
            'метод': 'Градиентный спуск',
            'начало': initial_angles,
            'конец': [round(a, 4) for a in angles],
            'целевой_угол': target,
            'финальный_ЛЗП': final_lci,
            'оптимальный_ЛЗП': optimal_lci,
            'сошёлся': all(abs(a - target) < 0.01 for a in angles),
        }

    # ДИНАМИЧЕСКОЕ ПРОГРАММИРОВАНИЕ: оптимальность ЧВС-цепочки
    @staticmethod
    def dynamic_programming_etd(
            cvs_sequence_length: int = 4,
            cvs_values: Dict[str, float] = None) -> dict:
        """
        Динамическое программирование для оптимальной ЧВС-цепочки:
        Найти последовательность ЧВС-состояний максимального суммарного ЛЗП.

        Принцип Беллмана:
        ЛЗП*(n) = max_{ЧВС_n} {r(ЧВС_n) + ЛЗП*(n-1)}.

        Оптимальная подзадача: ЛЗП*(k) не зависит от выбора после шага k.
        """
        if cvs_values is None:
            cvs_values = {
                'Внутреннее':      math.pi/4 * 1.0,
                'Инструментальное': math.pi/4 * 1.2,
                'Доменное':         math.pi/4 * 0.8,
                'Пространственное': math.pi/4 * 1.1,
                'Виртуальное':      math.pi/4 * 0.9,
            }

        # DP: на каждом шаге выбираем наилучшую ЧВС
        best_sequence = []
        best_lci = 0.0

        for step in range(cvs_sequence_length):
            best_cvs = max(cvs_values, key=cvs_values.get)
            best_sequence.append(best_cvs)
            best_lci += cvs_values[best_cvs]

        return {
            'метод': 'Динамическое программирование',
            'принцип': 'Беллман: ЛЗП*(n) = max_ЧВС {r(ЧВС) + ЛЗП*(n-1)}',
            'оптимальная_цепочка': best_sequence,
            'оптимальный_ЛЗП': best_lci,
            'вывод': 'ЧВС-цепочка: всегда выбирать наиболее ценное состояние',
        }

optim = OptimizationETD()

# Лагранж: π/4-синхронизация = оптимум!
lag_result = optim.lagrange_etd(constraint_energy=(math.pi/2)**2)
print(lag_result['вывод'])

# Градиентный спуск к π/4:
gd_result = optim.gradient_descent_etd([0.1, 0.5, 1.0, 1.5])
print(f"Градиентный спуск сошёлся: {gd_result['сошёлся']}")
```

## ЗАКЛЮЧЕНИЕ

**4 метода оптимизации** → **4 стратегии максимизации ЛЗП**:

1. **Лагранж** → При ограничении энергии: **оптимум = π/4-синхронизация** (МВС=СВС=БВС=ЧВС=π/4)
2. **Симплекс** → Линейная задача: оптимум — одна доминирующая сфера
3. **Градиентный спуск** → Итеративно: θₙ → π/4 (сходится к оптимуму)
4. **Динамическое программирование** → ЧВС-цепочка: принцип Беллмана

> **Теорема оптимальности**: При ограничении суммарной энергии Σθᵢ² = E, максимум ЛЗП достигается при МВС=СВС=БВС=ЧВС=π/4. Метод Лагранжа **доказывает**, что π/4-синхронизация есть глобальный оптимум.
